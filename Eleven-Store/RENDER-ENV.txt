Copy these keys into Render > Web Service > Environment:

NODE_ENV=production
ADMIN_PASSWORD=CHANGE_ME_ADMIN_PASSWORD
SESSION_SECRET=CHANGE_ME_LONG_RANDOM_SESSION_SECRET
BASE_URL=https://eleven-store-sa.onrender.com
GEIDEA_MERCHANT_PUBLIC_KEY=
GEIDEA_API_PASSWORD=
DISCORD_CLIENT_ID=CHANGE_ME_DISCORD_CLIENT_ID
DISCORD_CLIENT_SECRET=CHANGE_ME_DISCORD_CLIENT_SECRET
DISCORD_REDIRECT_URI=https://eleven-store-sa.onrender.com/auth/discord/callback

PORT is provided automatically by Render, so you normally do not need to add PORT there.

# Discord automatic Customer role after a completed purchase.
# Bot must be in the server, have Manage Roles, and its highest role must be above the Customer role.
DISCORD_BOT_TOKEN=CHANGE_ME_DISCORD_BOT_TOKEN
DISCORD_GUILD_ID=CHANGE_ME_DISCORD_GUILD_ID
DISCORD_CUSTOMER_ROLE_ID=CHANGE_ME_CUSTOMER_ROLE_ID

# Support assistant (per-chat can also be controlled from Admin)
SUPPORT_AI_ENABLED=true


# PayPal Checkout (use LIVE credentials for real payments)
PAYPAL_CLIENT_ID=YOUR_LIVE_CLIENT_ID
PAYPAL_CLIENT_SECRET=YOUR_LIVE_CLIENT_SECRET
PAYPAL_MODE=live
PAYPAL_CURRENCY=USD
PAYPAL_SAR_PER_UNIT=3.75

# Supabase persistence (keep these enabled on Render)
SUPABASE_URL=YOUR_SUPABASE_PROJECT_URL
SUPABASE_SERVICE_ROLE_KEY=YOUR_SUPABASE_SECRET_KEY
SUPABASE_BUCKET=eleven-store
SUPABASE_STATE_TABLE=store_state
